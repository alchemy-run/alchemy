export const BIND_SRC = `import * as Effect from "effect/Effect";
import * as DynamoDB from "alchemy/AWS/DynamoDB";
import * as SQS from "alchemy/AWS/SQS";
import { JobsTable, JobsQueue } from "./resources.ts";

export default class JobApi extends AWS.Lambda.Function<JobApi>()(
  "JobApi",
  Effect.gen(function* () {
    // Two binds — typed clients + IAM policies attached at deploy.
    const put  = yield* DynamoDB.PutItem.bind(JobsTable);
    const send = yield* SQS.SendMessage.bind(JobsQueue);

    return {
      fetch: Effect.gen(function* () {
        yield* put({ Item: { id: { S: "1" }, body: { S: "hi" } } });
        yield* send({ MessageBody: "1" });
      }),
    };
  }),
) {}`;

export const RETRY_SRC = `// All Effect operators work out of the box on bindings.
const sendWithRetry = send({ MessageBody: msg }).pipe(
  Effect.retry({ times: 3, schedule: Schedule.exponential("100 millis") }),
  Effect.timeout("5 seconds"),
  Effect.catchTag("ThrottlingException", () => Effect.succeed(undefined)),
);`;

export const LAYER_SRC = `// Bindings split into Service + Policy:
//   Binding.Service — the typed SDK wrapper, runs in your function
//   Binding.Policy  — the deploy-time IAM/env wiring, runs at plantime
//
// At runtime the Policy layer is absent, so it gracefully no-ops.
// Same code. Different layers. No conditionals.

const Live = Layer.mergeAll(
  DynamoDB.PutItemLive,
  SQS.SendMessageLive,
);`;
