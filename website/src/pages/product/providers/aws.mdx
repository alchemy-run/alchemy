export const STORAGE_SRC = `// JobStorage layer — DynamoDB table + SQS queue + DynamoDB stream
// fanning out into the queue, all wired with least-privilege IAM.
export const JobStorageDynamoDB = Layer.effect(
  JobStorage,
  Effect.gen(function* () {
    const table = yield* DynamoDB.Table("JobsTable", {
      partitionKey: "id",
      attributes: { id: "S" },
    });
    const queue = yield* SQS.Queue("JobsQueue");

    const get  = yield* DynamoDB.GetItem.bind(table);   // dynamodb:GetItem
    const put  = yield* DynamoDB.PutItem.bind(table);   // dynamodb:PutItem
    const sink = yield* SQS.QueueSink.bind(queue);      // sqs:SendMessage

    yield* DynamoDB.stream(table, {
      streamViewType: "NEW_AND_OLD_IMAGES",
      batchSize: 10,
    }).process((stream) =>                              // EventSourceMapping
      stream.pipe(
        Stream.map((rec) => JSON.stringify(rec)),
        Stream.run(sink),
      ),
    );

    return JobStorage.of({ get, put });
  }),
);`;
