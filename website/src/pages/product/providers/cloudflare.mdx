export const ASYNC_RUN_SRC = `// alchemy.run.ts — async style. Identical to v1.
import * as Alchemy from "alchemy";
import * as Cloudflare from "alchemy/Cloudflare";
import * as Effect from "effect/Effect";
import type { Counter } from "./src/worker.ts";

export const DB     = Cloudflare.D1Database("DB");
export const Bucket = Cloudflare.R2Bucket("Bucket");

export type WorkerEnv = Cloudflare.InferEnv<typeof Worker>;

export const Worker = Cloudflare.Worker("Worker", {
  main: "./src/worker.ts",
  bindings: {
    DB,
    Bucket,
    Counter: Cloudflare.DurableObjectNamespace<Counter>("Counter", {
      className: "Counter",
    }),
  },
});

export default Alchemy.Stack(
  "MyApp",
  { providers: Cloudflare.providers() },
  Effect.gen(function* () {
    const worker = yield* Worker;
    return worker.url;
  }),
);`;

export const ASYNC_WORKER_SRC = `// src/worker.ts — your standard async fetch handler.
// No Effect required. env is fully typed via InferEnv.
import { DurableObject } from "cloudflare:workers";
import type { WorkerEnv } from "../alchemy.run.ts";

export default {
  async fetch(request: Request, env: WorkerEnv) {
    const url = new URL(request.url);

    if (request.method === "GET") {
      return new Response((await env.Bucket.get(url.pathname))?.body ?? null);
    }
    if (request.method === "PUT") {
      await env.Bucket.put(url.pathname, request.body);
      return new Response(null, { status: 201 });
    }
    if (request.method === "POST") {
      const counter = env.Counter.getByName("counter");
      return new Response(String(await counter.increment()));
    }
    return env.ASSETS.fetch(request);
  },
};

export class Counter extends DurableObject {
  private counter = 0;
  async increment() { return ++this.counter; }
}`;

export const DO_SRC = `// Durable Objects with Hibernatable WebSockets.
// Idle sockets sleep without holding instance memory.
export default class Room extends Cloudflare.DurableObjectNamespace<Room>()(
  "Room",
  Effect.gen(function* () {
    return {
      fetch: function (this: Room, req) {
        const pair = new WebSocketPair();
        // Accept as a hibernatable socket — no JS heap held while idle.
        this.state.acceptWebSocket(pair[1]);
        return new Response(null, { status: 101, webSocket: pair[0] });
      },

      webSocketMessage: function (this: Room, ws, msg) {
        for (const peer of this.state.getWebSockets()) peer.send(msg);
      },
    };
  }),
) {}`;

export const CONTAINER_SRC = `// Bind a Container to a Durable Object — alchemy builds the
// image, starts it on demand, and gives you a typed .fetch handle.
export default class Agent extends Cloudflare.DurableObjectNamespace<Agent>()(
  "Agents",
  Effect.gen(function* () {
    const sandbox   = yield* Cloudflare.Container.bind(Sandbox);
    const container = yield* Cloudflare.start(sandbox);
    return { fetch: (req) => container.fetch(req) };
  }),
) {}`;

export const WORKFLOW_SRC = `// Durable workflows. Sleep, retry, resume across restarts.
export default class NotifyWorkflow extends Cloudflare.Workflow<NotifyWorkflow>()(
  "NotifyWorkflow",
  Effect.gen(function* () {
    const room = yield* Room;
    return {
      run: (event) => Effect.gen(function* () {
        yield* Effect.sleep("5 seconds");
        yield* room.getByName(event.payload.roomId)
                   .broadcast(event.payload.message);
      }),
    };
  }),
) {}`;
