export const STACK_SRC = `export default Alchemy.Stack(
  "MyApp",
  { providers: Cloudflare.providers() },
  Effect.gen(function* () {
    const bucket = yield* Bucket;
    const queue  = yield* Queue;
    const api    = yield* Api;

    // typed url returned to the caller
    return { url: api.url };
  }),
);`;

export const API_SRC = `export default class Api extends Cloudflare.Worker<Api>()(
  "Api",
  { main: import.meta.path, assets: "./assets" },
  Effect.gen(function* () {
    // typed runtime clients · bindings wired at deploy
    const bucket = yield* R2Bucket.bind(Bucket);
    const kv     = yield* KVNamespace.bind(KV);

    return { fetch: /* your handler uses bucket & kv */ };
  }),
) {}`;

export const BUCKET_SRC = `export const Bucket = Cloudflare.R2Bucket("Bucket");`;
export const KV_SRC = `export const KV = Cloudflare.KVNamespace("KV");`;

export const TEST_SRC = `// 1. deploy the real stack once, into an isolated stage
const stack = beforeAll(deploy(Stack, { stage: \`pr-\${Date.now()}\` }));
afterAll.skipIf(!process.env.CI)(destroy(Stack));

// 2. each test yields the live URL and hits it over HTTP
test("PUT + GET round-trips through R2", Effect.gen(function* () {
  const { url } = yield* stack;

  yield* HttpClient.put(\`\${url}/object/hello.txt\`, {
    body: HttpBody.text("hi!"),
  });

  const res = yield* HttpClient.get(\`\${url}/object/hello.txt\`);
  expect(yield* res.text).toBe("hi!");
}));

// 3. assert durable state — no mocks, just the live DO
test("Room DO preserves state across requests", Effect.gen(function* () {
  const { url } = yield* stack;
  yield* HttpClient.post(\`\${url}/room/general/join\`);
  const res = yield* HttpClient.get(\`\${url}/room/general\`);
  expect((yield* res.json).members).toHaveLength(1);
}));`;
