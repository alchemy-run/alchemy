## D1

```typescript
// src/Api.ts
import * as Cloudflare from "alchemy/Cloudflare";
import * as Drizzle from "alchemy/Drizzle/D1";
import { sql } from "drizzle-orm";
import * as Effect from "effect/Effect";
import * as HttpServerResponse from "effect/unstable/http/HttpServerResponse";

const Database = Cloudflare.D1.Database("Database", {
  migrations: "./migrations",
});

export default class Api extends Cloudflare.Worker<Api>()(
  "Api",
  { main: import.meta.url },
  Effect.gen(function* () {
    const connection = yield* Cloudflare.D1.QueryDatabase(Database);
    const db = yield* Drizzle.D1(connection);

    return {
      fetch: Effect.gen(function* () {
        const rows = yield* db.all(sql`SELECT 1 AS ok`);
        return yield* HttpServerResponse.json(rows);
      }).pipe(Effect.orDie),
    };
  }).pipe(Effect.provide(Cloudflare.D1.QueryDatabaseBinding)),
) {}
```

## Postgres and MySQL

A Worker with Neon Postgres through Hyperdrive:

```typescript
// src/PostgresApi.ts
import * as Cloudflare from "alchemy/Cloudflare";
import * as Drizzle from "alchemy/Drizzle/Postgres";
import * as Neon from "alchemy/Neon";
import { sql } from "drizzle-orm";
import * as Effect from "effect/Effect";
import * as HttpServerResponse from "effect/unstable/http/HttpServerResponse";

const Hyperdrive = Effect.gen(function* () {
  const project = yield* Neon.Project("Database");
  const branch = yield* Neon.Branch("App", {
    project,
    migrations: "./migrations",
  });
  return yield* Cloudflare.Hyperdrive.Connection("Hyperdrive", {
    origin: branch.origin,
    caching: { disabled: true },
  });
});

export default class Api extends Cloudflare.Worker<Api>()(
  "Api",
  { main: import.meta.url, compatibility: { flags: ["nodejs_compat"] } },
  Effect.gen(function* () {
    const connection = yield* Cloudflare.Hyperdrive.Connect(Hyperdrive);
    const db = yield* Drizzle.Postgres(connection.connectionString);

    return {
      fetch: Effect.gen(function* () {
        const result = yield* db.execute(sql`SELECT 1 AS ok`);
        return yield* HttpServerResponse.json(result);
      }).pipe(Effect.orDie),
    };
  }).pipe(Effect.provide(Cloudflare.Hyperdrive.ConnectBinding)),
) {}
```
