## Prisma Postgres

```typescript
// src/database.ts
import * as Prisma from "alchemy/Prisma";
import * as Effect from "effect/Effect";

export const Project = Prisma.Project("App", {
  createDatabase: false,
  region: "us-east-1",
});

export const Connection = Effect.gen(function* () {
  const project = yield* Project;
  const database = yield* Prisma.Postgres("Database", { project });
  return yield* Prisma.Connection("Connection", { database });
});
```

## Prisma Compute

```typescript
// src/Api.ts
import * as Drizzle from "alchemy/Drizzle/Postgres";
import * as Prisma from "alchemy/Prisma";
import { sql } from "drizzle-orm";
import * as Effect from "effect/Effect";
import * as HttpServerResponse from "effect/unstable/http/HttpServerResponse";
import { Connection, Project } from "./database.ts";

export default class Api extends Prisma.Compute<Api>()(
  "Api",
  Effect.gen(function* () {
    const project = yield* Project;
    return { project, main: import.meta.filename };
  }),
  Effect.gen(function* () {
    const connection = yield* Prisma.Connect(Connection);
    const db = yield* Drizzle.Postgres(connection.databaseUrl);

    return {
      fetch: Effect.gen(function* () {
        const result = yield* db.execute(sql`SELECT 1 AS ok`);
        return yield* HttpServerResponse.json(result);
      }).pipe(Effect.orDie),
    };
  }).pipe(Effect.provide(Prisma.ConnectBinding)),
) {}
```

## Prisma ORM

```typescript
// src/migrations.ts
import * as Prisma from "alchemy/Prisma";
import * as Config from "effect/Config";
import * as Effect from "effect/Effect";

export const Migrations = Effect.gen(function* () {
  const contract = yield* Prisma.Contract("Contract");
  return yield* Prisma.Migrate("Migrate", {
    url: yield* Config.Redacted("DATABASE_URL"),
    contract,
  });
});
```

Yield `Migrations` from the Stack. Prisma ORM is separate from Prisma hosting; ORM v8 requires PostgreSQL 17+.
