## Postgres

```typescript
// src/Api.ts
import * as Drizzle from "alchemy/Drizzle/Postgres";
import * as Railway from "alchemy/Railway";
import { sql } from "drizzle-orm";
import * as Effect from "effect/Effect";
import * as HttpServerResponse from "effect/unstable/http/HttpServerResponse";

const Project = Railway.Project("App");
const Database = Railway.Postgres("Database", { project: Project });

export default class Api extends Railway.Service<Api>()(
  "Api",
  {
    project: Project,
    main: import.meta.url,
    build: { install: ["pg"] },
  },
  Effect.gen(function* () {
    const connection = yield* Railway.ConnectPostgres(Database);
    const db = yield* Drizzle.Postgres(connection.connectionString);

    return {
      fetch: Effect.gen(function* () {
        const result = yield* db.execute(sql`SELECT 1 AS ok`);
        return yield* HttpServerResponse.json(result);
      }).pipe(Effect.orDie),
    };
  }).pipe(Effect.provide(Railway.ConnectPostgresHttp)),
) {}
```

Private hostnames are reachable inside Railway; external migration runners need the public endpoint.

## MySQL

```typescript
// src/MySqlApi.ts
import * as Drizzle from "alchemy/Drizzle/MySQL";
import * as Railway from "alchemy/Railway";
import { sql } from "drizzle-orm";
import * as Effect from "effect/Effect";
import * as HttpServerResponse from "effect/unstable/http/HttpServerResponse";

const Project = Railway.Project("App");
const Database = Railway.MySQL("Database", { project: Project });

export default class Api extends Railway.Service<Api>()(
  "Api",
  {
    project: Project,
    main: import.meta.url,
    build: { install: ["mysql2"] },
  },
  Effect.gen(function* () {
    const connection = yield* Railway.ConnectMySQL(Database);
    const db = yield* Drizzle.MySQL(connection.connectionString);

    return {
      fetch: Effect.gen(function* () {
        const result = yield* db.execute(sql`SELECT 1 AS ok`, "objects");
        return yield* HttpServerResponse.json(result);
      }).pipe(Effect.orDie),
    };
  }).pipe(Effect.provide(Railway.ConnectMySQLHttp)),
) {}
```
