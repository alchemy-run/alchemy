## Postgres

```typescript
// src/Api.ts
import * as Cloudflare from "alchemy/Cloudflare";
import * as Drizzle from "alchemy/Drizzle/Postgres";
import * as Planetscale from "alchemy/Planetscale";
import { sql } from "drizzle-orm";
import * as Effect from "effect/Effect";
import * as HttpServerResponse from "effect/unstable/http/HttpServerResponse";

const Hyperdrive = Effect.gen(function* () {
  const database = yield* Planetscale.PostgresDatabase("Database", {
    region: { slug: "us-east" },
    clusterSize: "PS_10",
  });
  const branch = yield* Planetscale.PostgresBranch("App", {
    database,
    migrations: "./migrations",
  });
  const role = yield* Planetscale.PostgresRole("Reader", {
    database,
    branch,
    inheritedRoles: ["pg_read_all_data"],
  });
  return yield* Cloudflare.Hyperdrive.Connection("Hyperdrive", {
    origin: role.origin,
    caching: { disabled: true },
  });
});

export default class Api extends Cloudflare.Worker<Api>()(
  "Api",
  { main: import.meta.url, compatibility: { flags: ["nodejs_compat"] } },
  Effect.gen(function* () {
    const connection = yield* Cloudflare.Hyperdrive.Connect(Hyperdrive);
    const db = yield* Drizzle.Postgres(connection.connectionString);

    return {
      fetch: Effect.gen(function* () {
        const result = yield* db.execute(sql`SELECT 1 AS ok`);
        return yield* HttpServerResponse.json(result);
      }).pipe(Effect.orDie),
    };
  }).pipe(Effect.provide(Cloudflare.Hyperdrive.ConnectBinding)),
) {}
```

## MySQL

```typescript
// src/MySqlApi.ts
import * as Cloudflare from "alchemy/Cloudflare";
import * as Drizzle from "alchemy/Drizzle/MySQL";
import * as Planetscale from "alchemy/Planetscale";
import { sql } from "drizzle-orm";
import * as Effect from "effect/Effect";
import * as HttpServerResponse from "effect/unstable/http/HttpServerResponse";

const Hyperdrive = Effect.gen(function* () {
  const database = yield* Planetscale.MySQLDatabase("Database", {
    region: { slug: "us-east" },
    clusterSize: "PS_10",
  });
  const branch = yield* Planetscale.MySQLBranch("App", {
    database,
    isProduction: false,
    migrations: "./migrations",
  });
  const password = yield* Planetscale.MySQLPassword("Reader", {
    database,
    branch,
    role: "reader",
  });
  return yield* Cloudflare.Hyperdrive.Connection("Hyperdrive", {
    origin: password.origin,
    caching: { disabled: true },
  });
});

export default class Api extends Cloudflare.Worker<Api>()(
  "Api",
  { main: import.meta.url, compatibility: { flags: ["nodejs_compat"] } },
  Effect.gen(function* () {
    const connection = yield* Cloudflare.Hyperdrive.Connect(Hyperdrive);
    const db = yield* Drizzle.MySQL(connection.connectionString);

    return {
      fetch: Effect.gen(function* () {
        const result = yield* db.execute(sql`SELECT 1 AS ok`, "objects");
        return yield* HttpServerResponse.json(result);
      }).pipe(Effect.orDie),
    };
  }).pipe(Effect.provide(Cloudflare.Hyperdrive.ConnectBinding)),
) {}
```
