## Fly Managed Postgres

```typescript
// src/Api.ts
import * as Drizzle from "alchemy/Drizzle/Postgres";
import * as Fly from "alchemy/Fly";
import { sql } from "drizzle-orm";
import * as Effect from "effect/Effect";
import * as HttpServerResponse from "effect/unstable/http/HttpServerResponse";

const App = Fly.App("App");
const Database = Fly.Postgres("Database", {
  region: "iad",
  migrations: "./migrations",
});

export default class Api extends Fly.Service<Api>()(
  "Api",
  {
    app: App,
    main: import.meta.url,
    build: { install: ["pg"] },
  },
  Effect.gen(function* () {
    const connection = yield* Fly.ConnectPostgres(Database);
    const db = yield* Drizzle.Postgres(connection.connectionString);

    return {
      fetch: Effect.gen(function* () {
        const result = yield* db.execute(sql`SELECT 1 AS ok`);
        return yield* HttpServerResponse.json(result);
      }).pipe(Effect.orDie),
    };
  }).pipe(Effect.provide(Fly.ConnectPostgresHttp)),
) {}
```

Deployment-time migrations need access to Fly's private network.
