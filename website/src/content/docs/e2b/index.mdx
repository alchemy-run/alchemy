E2B runs sandboxes: Linux microVMs that start in well under a second from a template you build, run untrusted code (an agent's commands, a user's script) with no route into your systems, and can be paused, resumed, snapshotted and forked. Declare templates, snapshots, volumes and secrets in your Stack, and create sandboxes from any Worker or Function.

**New to E2B?** Start with [What is E2B?](/e2b/what-is-e2b).

## A quick example

A template is the image every sandbox boots from:

```typescript title="src/Sandbox.ts"
import * as Dockerfile from "@alchemy.run/docker/Dockerfile";
import * as E2B from "@alchemy.run/e2b";

export const Agent = E2B.Template("Agent", {
  dockerfile: Dockerfile.inline`
    FROM node:22-bookworm
    RUN apt-get update && apt-get install -y git ripgrep
  `,
  cpuCount: 2,
  memoryMB: 2048,
});
```

A Worker creates a sandbox from it on each request, runs the code it was sent, and kills the sandbox when the request ends:

```typescript title="src/Api.ts"
import * as Cloudflare from "@alchemy.run/cloudflare";
import * as E2B from "@alchemy.run/e2b";
import * as Effect from "effect/Effect";
import { HttpServerRequest } from "effect/http/HttpServerRequest";
import * as HttpServerResponse from "effect/http/HttpServerResponse";
import { Agent } from "./Sandbox.ts";

export default class Api extends Cloudflare.Worker<Api>()(
  "Api",
  { main: import.meta.url },
  Effect.gen(function* () {
    const sandboxes = yield* E2B.CreateSandbox(Agent);
    return {
      fetch: Effect.gen(function* () {
        const request = yield* HttpServerRequest;
        const sandbox = yield* Effect.acquireRelease(
          sandboxes.create({ timeout: 120, network: { allowOut: [] } }),
          (sandbox) => Effect.ignore(sandbox.kill()),
        );
        yield* sandbox.files.write("main.mjs", yield* request.text);
        const result = yield* sandbox.exec("node main.mjs");
        return yield* HttpServerResponse.json(result);
      }).pipe(Effect.scoped, Effect.orDie),
    };
  }).pipe(Effect.provide(E2B.CreateSandboxHttp)),
) {}
```

And the Stack deploys both:

```typescript title="alchemy.run.ts"
import * as Cloudflare from "@alchemy.run/cloudflare";
import * as E2B from "@alchemy.run/e2b";
import * as Alchemy from "alchemy";
import * as Effect from "effect/Effect";
import * as Layer from "effect/Layer";
import Api from "./src/Api.ts";
import { Agent } from "./src/Sandbox.ts";

export default Alchemy.Stack(
  "MyApp",
  {
    providers: Layer.mergeAll(Cloudflare.providers(), E2B.providers()),
    state: Cloudflare.state(),
  },
  Effect.gen(function* () {
    const agent = yield* Agent;
    const api = yield* Api;
    return { url: api.url.as<string>(), templateId: agent.templateId };
  }),
);
```
