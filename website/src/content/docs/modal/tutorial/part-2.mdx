## Bake the repo into a warm base

```diff lang="typescript"
// src/Sandbox.ts
export const Node = Modal.Image("Node", {
  app: Coder,
  image: "node:22-bookworm",
  commands: ["RUN apt-get update && apt-get install -y git"],
});
+
+export const Warm = Modal.Image("Warm", {
+  app: Coder,
+  from: Node,
+  run: ["git clone --depth 1 https://github.com/sindresorhus/is /work"],
+});
```

`run` executes in a sandbox once at deploy, then saves the sandbox's files as an image.

## Bind sandboxes and volumes in a Worker

```typescript
// src/Api.ts
import * as Cloudflare from "@alchemy.run/cloudflare";
import * as Modal from "@alchemy.run/modal";
import * as Effect from "effect/Effect";
import * as Layer from "effect/Layer";
import * as HttpServerResponse from "effect/http/HttpServerResponse";
import { Agents, Warm } from "./Sandbox.ts";

export default class Api extends Cloudflare.Worker<Api>()(
  "Api",
  { main: import.meta.url, compatibility: { flags: ["nodejs_compat"] } },
  Effect.gen(function* () {
    const sandboxes = yield* Modal.CreateSandbox(Warm);
    const volumes = yield* Modal.CreateVolume(Agents);
    return {
      fetch: Effect.succeed(HttpServerResponse.text("ok")),
    };
  }).pipe(Effect.provide(Layer.mergeAll(Modal.CreateSandboxHttp, Modal.CreateVolumeHttp))),
) {}
```

`CreateSandbox(Warm)` boots sandboxes from the warm base. `CreateVolume(Agents)` creates volumes by name in the environment.

## Save a user's notes

```diff lang="typescript"
// src/Api.ts
-      fetch: Effect.succeed(HttpServerResponse.text("ok")),
+      fetch: Effect.gen(function* () {
+        const request = yield* HttpServerRequest;
+        const url = new URL(request.url, "http://api");
+        const user = url.searchParams.get("user") ?? "anonymous";
+        const home = yield* volumes.getOrCreate(`home-${user}`);
+        if (request.method === "PUT") {
+          yield* home.writeFile("/notes.txt", yield* request.text);
+        }
+        const notes = yield* home.readFileString("/notes.txt").pipe(
+          Effect.catchTag("VolumeFileNotFound", () => Effect.succeed("")),
+        );
+        return HttpServerResponse.text(notes);
+      }).pipe(Effect.orDie),
```

The Worker writes the file straight to the volume. No sandbox runs.

## Read them from a sandbox

```diff lang="typescript"
// src/Api.ts
+        if (url.pathname === "/run") {
+          const sandbox = yield* sandboxes.create({
+            mounts: [{ volume: home, path: "/home/user" }],
+            timeout: "5 minutes",
+          });
+          const result = yield* sandbox
+            .exec("ls /work && cat /home/user/notes.txt")
+            .pipe(Effect.ensuring(Effect.ignore(sandbox.terminate())));
+          return yield* HttpServerResponse.json(result);
+        }
```

The sandbox mounts the user's volume at `/home/user` and stops when the request is done.

## Try it

```sh
curl -X PUT "$URL/notes?user=ada" -d "hello from ada"
curl "$URL/run?user=ada"
```

The second call prints the repo's files and `hello from ada`.
