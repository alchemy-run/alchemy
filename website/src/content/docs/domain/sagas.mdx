---
title: Sagas and compensation
description: Move a process across several aggregates in steps, record where it got to, and undo a step when a later one fails.
---

A transfer debits one account and credits another. No transaction spans two accounts, so the transfer runs as a saga: a sequence of steps, each committed on its own, with the process's own state recording how far it got. When a later step fails, an earlier one is undone with a compensating step.

## Keep the saga's state in an aggregate

The transfer is an aggregate. Its state is where the process stands: waiting for approval, pending, completed or failed.

```typescript
// src/Transfer/Transfer.ts
import * as Domain from "alchemy/Domain";
import * as Effect from "effect/Effect";
import * as Schema from "effect/Schema";
import { AccountId } from "../Account/AccountId.ts";
import { NotPermitted } from "../Account/ChangeOwner.ts";
import { Actor } from "../Actor.ts";
import { CurrentActor } from "../CurrentActor.ts";
import { Cents } from "../Money.ts";
import { ApproveTransfer, NotAwaitingApproval, TransferApproved } from "./ApproveTransfer.ts";
import { CompleteTransfer, TransferCompleted } from "./CompleteTransfer.ts";
import { ExpireTransfer } from "./ExpireTransfer.ts";
import { FailTransfer, TransferFailed } from "./FailTransfer.ts";
import {
  AlreadyRequested,
  APPROVAL_THRESHOLD,
  RequestTransfer,
  SameAccount,
  TransferRequested,
} from "./RequestTransfer.ts";
import { TransferId } from "./TransferId.ts";

const Details = { from: AccountId, to: AccountId, amount: Cents, requestedBy: Actor };

/**
 * A transfer between two accounts: the state of the TransferExecution
 * process. Large transfers wait for approval until their window closes;
 * approval and expiry race on this single instance, so one of them wins.
 */
export class Transfer extends Domain.Aggregate("Transfer", {
  id: TransferId,
  state: Schema.Union([
    Schema.TaggedStruct("New", {}),
    Schema.TaggedStruct("AwaitingApproval", Details),
    Schema.TaggedStruct("Pending", Details),
    Schema.TaggedStruct("Completed", {}),
    Schema.TaggedStruct("Failed", { reason: Schema.String }),
  ]),
  initial: { _tag: "New" },
  commands: [RequestTransfer, ApproveTransfer, ExpireTransfer, CompleteTransfer, FailTransfer],
  decide: {
    // Records who asked, so TransferExecution debits the source on their behalf.
    RequestTransfer: Effect.fn(function* (s, cmd, { id }) {
      if (s._tag !== "New") return yield* new AlreadyRequested();
      if (cmd.from === cmd.to) return yield* new SameAccount();
      const requestedBy = yield* CurrentActor;
      return [
        new TransferRequested({
          transferId: id,
          from: cmd.from,
          to: cmd.to,
          amount: cmd.amount,
          requestedBy,
          needsApproval: cmd.amount >= APPROVAL_THRESHOLD,
        }),
      ];
    }),
    ApproveTransfer: Effect.fn(function* (s, _, { id }) {
      const approvedBy = yield* CurrentActor;
      if (approvedBy._tag !== "Agent") return yield* new NotPermitted();
      if (s._tag !== "AwaitingApproval") return yield* new NotAwaitingApproval();
      return [
        new TransferApproved({
          transferId: id,
          from: s.from,
          to: s.to,
          amount: s.amount,
          requestedBy: s.requestedBy,
          approvedBy,
        }),
      ];
    }),
    ExpireTransfer: (s, _, { id }) =>
      s._tag === "AwaitingApproval"
        ? [new TransferFailed({ transferId: id, reason: "not approved in time" })]
        : [],
    CompleteTransfer: (s, _, { id }) =>
      s._tag === "Pending" ? [new TransferCompleted({ transferId: id })] : [],
    FailTransfer: (s, cmd, { id }) =>
      s._tag === "Pending" ? [new TransferFailed({ transferId: id, reason: cmd.reason })] : [],
  },
  evolve: {
    TransferRequested: (_, e) => ({
      _tag: e.needsApproval ? ("AwaitingApproval" as const) : ("Pending" as const),
      from: e.from,
      to: e.to,
      amount: e.amount,
      requestedBy: e.requestedBy,
    }),
    TransferApproved: (_, e) => ({
      _tag: "Pending" as const,
      from: e.from,
      to: e.to,
      amount: e.amount,
      requestedBy: e.requestedBy,
    }),
    TransferCompleted: () => ({ _tag: "Completed" as const }),
    TransferFailed: (_, e) => ({ _tag: "Failed" as const, reason: e.reason }),
  },
}) {}
```

Every in-between state is part of the model, so anyone can see a transfer that's still pending, and support can approve one that's waiting.

## Drive it with a policy

A policy reacts to the transfer's events and carries out each step on the accounts.

```typescript
// src/Transfer/TransferExecution.ts
import * as Domain from "alchemy/Domain";
import * as Effect from "effect/Effect";
import { Account } from "../Account/Account.ts";
import { CreditTransfer } from "../Account/CreditTransfer.ts";
import { DebitTransfer } from "../Account/DebitTransfer.ts";
import { RefundTransfer } from "../Account/RefundTransfer.ts";
import { TransferApproved } from "./ApproveTransfer.ts";
import { CompleteTransfer } from "./CompleteTransfer.ts";
import { FailTransfer } from "./FailTransfer.ts";
import { TransferRequested } from "./RequestTransfer.ts";
import { Transfer } from "./Transfer.ts";

/**
 * Drives a transfer once it may run (requested, or approved if it needed
 * approval): debit the source on behalf of the requester, credit the
 * destination, then complete, or compensate and fail.
 */
export class TransferExecution extends Domain.Policy("TransferExecution", {
  from: Transfer,
  on: [TransferRequested, TransferApproved],
}) {}

export const TransferExecutionLive = TransferExecution.toLayer(
  Effect.gen(function* () {
    const accounts = yield* Account;
    const transfers = yield* Transfer;
    return Effect.fn(function* ({ source, event }) {
      if (event._tag === "TransferRequested" && event.needsApproval) return;
      const { transferId, from, to, amount, requestedBy } = event;
      const fail = (reason: string) =>
        transfers.send(source, new FailTransfer({ reason })).pipe(Effect.as(false));

      // Account.decide checks that the requester owns the source account.
      const debited = yield* accounts
        .send(from, new DebitTransfer({ transferId, amount, requestedBy }))
        .pipe(
          Effect.as(true),
          Effect.catchTags({
            NotOpen: () => fail("source account not open"),
            NotOwner: () => fail("not your account"),
            Frozen: (r) => fail(`source frozen: ${r.reason}`),
            InsufficientFunds: () => fail("insufficient funds"),
          }),
        );
      if (!debited) return;

      yield* accounts.send(to, new CreditTransfer({ transferId, amount })).pipe(
        Effect.andThen(transfers.send(source, new CompleteTransfer())),
        Effect.asVoid,
        Effect.catchTag("NotOpen", () =>
          accounts
            .send(from, new RefundTransfer({ transferId, amount }))
            .pipe(Effect.andThen(fail("destination account not open")), Effect.asVoid),
        ),
      );
    });
  }),
);
```

Each step is a command to one aggregate. A step that fails comes back as a typed rejection, and the policy records the outcome on the transfer with `FailTransfer`.

## Take the step you can't undo first

Debit before you credit. If the credit then fails, the money is refunded, so it can be in flight but never created out of nothing. In the same way, registration claims the username before the customer counts as registered: see [Unique values](/domain/unique-values).

## Compensate a failed step

When the destination account isn't open, the policy refunds the debit and fails the transfer. Commands that report an outcome, like `CompleteTransfer` and `FailTransfer`, never reject, so recording the outcome can't fail.

```typescript
// src/Transfer/FailTransfer.ts
import * as Domain from "alchemy/Domain";
import * as Schema from "effect/Schema";
import { TransferId } from "./TransferId.ts";

export class TransferFailed extends Domain.Event("TransferFailed", {
  transferId: TransferId,
  reason: Schema.String,
}) {}

/** Reports an outcome, so it never rejects. */
export class FailTransfer extends Domain.Command("FailTransfer", {
  target: TransferId,
  input: { reason: Schema.String },
  events: [TransferFailed],
}) {}
```

## Follow it

`TransferStatus` is a view of the transfer and both accounts. A client watches it to see the saga finish.

```typescript
// test/followTransfer.ts
import * as Effect from "effect/Effect";
import * as Stream from "effect/Stream";
import { TransferId } from "../src/Transfer/TransferId.ts";
import { TransferStatus } from "../src/Transfer/TransferStatus.ts";
import type { CustomerClient } from "./flows.ts";

export const followTransfer = Effect.fn(function* (bank: CustomerClient, transferId: string) {
  yield* bank.TransferStatus.watch(TransferId.make(transferId)).pipe(
    Stream.takeUntil((status) => status?.completedAt !== null || status?.failedReason !== null),
    Stream.runForEach((status) =>
      Effect.log(status === null ? "pending" : TransferStatus.computed.status(status)),
    ),
  );
});
```

## Test it

```typescript
// test/transfers.test.ts
import { it } from "@effect/vitest";
import { Story } from "alchemy/Domain/Story";
import { Account } from "../src/Account/Account.ts";
import { AccountClosed } from "../src/Account/CloseAccount.ts";
import { TransferCredited } from "../src/Account/CreditTransfer.ts";
import { TransferDebited } from "../src/Account/DebitTransfer.ts";
import { MoneyDeposited } from "../src/Account/DepositMoney.ts";
import { AccountOpened } from "../src/Account/OpenAccount.ts";
import { TransferRefunded } from "../src/Account/RefundTransfer.ts";
import { customer } from "../src/Actor.ts";
import * as Layer from "effect/Layer";
import { Bank, BankHandlers } from "../src/Bank.ts";
import { MemoryLedgerStore } from "../src/Ledger/LedgerStore.ts";
import { CurrentActor } from "../src/CurrentActor.ts";
import { CustomerId } from "../src/Customer/CustomerId.ts";
import { FraudCheck } from "../src/Fraud/FraudCheck.ts";
import { Payments } from "../src/Settlement/Payments.ts";
import { TransferCompleted } from "../src/Transfer/CompleteTransfer.ts";
import { TransferFailed } from "../src/Transfer/FailTransfer.ts";
import { RequestTransfer, TransferRequested } from "../src/Transfer/RequestTransfer.ts";
import { Transfer } from "../src/Transfer/Transfer.ts";

const story = Story.make(Bank, {
  layer: BankHandlers.pipe(Layer.provide(MemoryLedgerStore)),
  externalSystems: [FraudCheck, Payments],
});
const sam = customer("c-1");
const from = Account.ref("a-1");
const to = Account.ref("a-2");
const transfer = Transfer.ref("t-1");
const opened = (customerId: string) =>
  new AccountOpened({ customerId: CustomerId.make(customerId), owner: customerId });
const requested = new TransferRequested({
  transferId: transfer.id,
  from: from.id,
  to: to.id,
  amount: 40,
  requestedBy: sam,
  needsApproval: false,
});

it.effect("a transfer debits, credits, then completes", () =>
  story(
    Story.given(from, opened("c-1"), new MoneyDeposited({ amount: 100, balanceAfter: 100 })),
    Story.given(to, opened("c-2")),
    Story.provide(CurrentActor, sam),
    Story.when(transfer, new RequestTransfer({ from: from.id, to: to.id, amount: 40 })),
    Story.then(transfer, requested),
    Story.then(from, new TransferDebited({ transferId: transfer.id, amount: 40, balanceAfter: 60 })),
    Story.then(to, new TransferCredited({ transferId: transfer.id, amount: 40, balanceAfter: 40 })),
    Story.then(transfer, new TransferCompleted({ transferId: transfer.id })),
  ),
);

it.effect("a transfer to a closed account is refunded and fails", () =>
  story(
    Story.given(from, opened("c-1"), new MoneyDeposited({ amount: 100, balanceAfter: 100 })),
    Story.given(to, opened("c-2"), new AccountClosed()),
    Story.provide(CurrentActor, sam),
    Story.when(transfer, new RequestTransfer({ from: from.id, to: to.id, amount: 40 })),
    Story.then(transfer, requested),
    Story.then(
      from,
      new TransferDebited({ transferId: transfer.id, amount: 40, balanceAfter: 60 }),
      new TransferRefunded({ transferId: transfer.id, amount: 40, balanceAfter: 100 }),
    ),
    Story.then(
      transfer,
      new TransferFailed({ transferId: transfer.id, reason: "destination account not open" }),
    ),
  ),
);
```
